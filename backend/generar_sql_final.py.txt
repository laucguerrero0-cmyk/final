import pymysql
from mascotas_microservicios.common import config
import os

# Ruta donde está tu config
print("Conectando a la base...")

conn = pymysql.connect(
    host=config.DB_HOST,
    user=config.DB_USER,
    password=config.DB_PASSWORD,
    database=config.DB_NAME,
    charset="utf8mb4"
)

cur = conn.cursor()

ruta_salida = "mascotas_microservicios/base_de_datos/mascotas_rest_db_FINAL.sql"
# Crear carpeta si no existe
os.makedirs(os.path.dirname(ruta_salida), exist_ok=True)

with open(ruta_salida, "w", encoding="utf-8") as f:
    f.write(f"-- BASE REAL: {config.DB_NAME}\n")
    f.write(f"-- FINAL PARA ENTREGA\n\n")
    f.write("SET FOREIGN_KEY_CHECKS=0;\n\n")

    cur.execute("SHOW TABLES")
    tablas = [row[0] for row in cur.fetchall()]
    print(f"Tablas encontradas: {tablas}")

    for tabla in tablas:
        print(f"Exportando {tabla}...")
        f.write(f"DROP TABLE IF EXISTS `{tabla}`;\n")
        cur.execute(f"SHOW CREATE TABLE `{tabla}`")
        create = cur.fetchone()[1]
        f.write(f"{create};\n\n")

        cur.execute(f"SELECT * FROM `{tabla}`")
        filas = cur.fetchall()
        columnas = [desc[0] for desc in cur.description]

        for fila in filas:
            valores = []
            for val in fila:
                if val is None:
                    valores.append("NULL")
                elif isinstance(val, (int, float)):
                    valores.append(str(val))
                else:
                    v = str(val).replace("'", "''")
                    valores.append(f"'{v}'")
            cols = "`, `".join(columnas)
            vals = ", ".join(valores)
            f.write(f"INSERT INTO `{tabla}` (`{cols}`) VALUES ({vals});\n")
        f.write("\n\n")

    f.write("SET FOREIGN_KEY_CHECKS=1;\n")

print(f"¡LISTO! Archivo creado en {ruta_salida}")
cur.close()
conn.close()
